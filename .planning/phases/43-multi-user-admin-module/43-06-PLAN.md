"""Phase 43 users module router (USERS-01..05 — D-43-09 / D-43-29).

Dependency-order discipline (RBAC-04 / clients/router.py:16-19):
  1. body / path / query parameters first
  2. require_permission (401 then 403)
  3. verify_csrf (403 on missing/invalid token — mutations only)
  4. get_db (session)

GET /api/v1/users omits verify_csrf (CSRF on mutations only).

Action verbs (D-41-22 / D-43-11):

  - POST / → Action.CREATE

  - DELETE /{id} → Action.DELETE
"""

@router.get(
    "",
    response_model=ResponseEnvelope[PaginatedData[UserListItemResponse]],
)
async def list_users_endpoint(
    query: Annotated[UserListQuery, Depends()],
    _actor: Annotated[CurrentUser, Depends(require_permission(Action.LIST, Resource.USERS))],
    session: Annotated[AsyncSession, Depends(get_db)],
) -> ResponseEnvelope[PaginatedData[UserListItemResponse]]:
    """USERS-02 — paginated list with optional filters."""
    data = await service.list_users(session, query)
    return envelope(data)


@router.post(
    "",
    response_model=ResponseEnvelope[UserCreateResponse],
    status_code=status.HTTP_201_CREATED,
)
async def create_user_endpoint(
    payload: UserCreateRequest,
    actor: Annotated[CurrentUser, Depends(require_permission(Action.CREATE, Resource.USERS))],
    _csrf: Annotated[None, Depends(verify_csrf)],
    session: Annotated[AsyncSession, Depends(get_db)],
    include_invite_link: Annotated[bool, Query(alias="include_invite_link")] = False,
) -> ResponseEnvelope[UserCreateResponse]:
    """USERS-03 — create + invite (D-43-13). Idempotent on pending re-invite."""
    try:
        result = await service.create_user(session, actor, payload, include_invite_link)
    except service.EmailAlreadyActiveError as exc:
        raise HTTPException(status_code=409, detail={"error": "email_already_active"}) from exc
    return envelope(result)


@router.patch(
    "/{user_id}/deactivate",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def deactivate_user_endpoint(
    user_id: UUID,
    actor: Annotated[CurrentUser, Depends(require_permission(Action.UPDATE, Resource.USERS))],
    _csrf: Annotated[None, Depends(verify_csrf)],
    session: Annotated[AsyncSession, Depends(get_db)],
) -> None:
    """USERS-04 / D-43-16 — atomic deactivate + session revoke."""
    try:
        await service.deactivate_user(session, actor, user_id)
    except service.UserNotFoundError as exc:
        raise HTTPException(status_code=404, detail={"error": "user_not_found"}) from exc
    except service.UserAlreadyInactiveError as exc:
        raise HTTPException(status_code=409, detail={"error": "user_already_inactive"}) from exc
    except service.CannotDeactivateSelfError as exc:
        raise HTTPException(status_code=409, detail={"error": "cannot_deactivate_self"}) from exc
    except service.CannotDeactivateLastOwnerError as exc:
        raise HTTPException(status_code=409, detail={"error": "cannot_deactivate_last_owner"}) from exc
