"""Phase 43 USERS-03 invitation flow — D-43-13/14/19/25.

Sandbox email capture (D-42-03 / Phase 42 plan 07 — SandboxEmailClient stub).
"""
from __future__ import annotations

from uuid import UUID

import pytest
from httpx import AsyncClient
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.audit_models import AuditLog
from app.integrations.email.client import SandboxEmailClient
from app.modules.auth.password_reset_token_model import PasswordResetToken

pytestmark = pytest.mark.asyncio


def _csrf(client: AsyncClient) -> dict[str, str]:
    return {"X-CSRF-Token": client.cookies.get("sportzal_csrf", "")}


async def test_invitation_email_enqueued_through_sandbox(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
    sandbox_email_client: SandboxEmailClient,  # fixture: the SandboxEmailClient instance the test app is wired to
) -> None:
    """D-43-13/24 — POST /users enqueues USER_INVITATION_EMAIL with rendered envelope."""
    sandbox_email_client.sent_emails.clear()  # or equivalent reset method

    r = await authed_client_owner.post(
        "/api/v1/users",
        json={"email": "invitee@example.com", "fullName": "Иван Тестовый", "role": "reception"},
        headers=_csrf(authed_client_owner),
    )
    assert r.status_code == 201, r.text

    # Sandbox captured exactly one email targeted at the invitee.
    sent = sandbox_email_client.sent_emails
    assert len(sent) == 1, f"expected 1 sandbox email; got {len(sent)}"
    email = sent[0]
    assert email["to"] == "invitee@example.com"
    assert email["template_id"] == "USER_INVITATION_EMAIL"
    assert "Приглашение в Sportzal" in email["subject"]
    assert "Иван Тестовый" in email["text"]
    assert "/auth/accept-invite#token=" in email["text"]


async def test_include_invite_link_returns_url_with_fragment(
    authed_client_owner: AsyncClient,
) -> None:
    """D-43-14 — ?include_invite_link=true exposes the URL with #token= fragment for copy-paste."""
    r = await authed_client_owner.post(
        "/api/v1/users?include_invite_link=true",
        json={"email": "with-link2@example.com", "fullName": "X", "role": "reception"},
        headers=_csrf(authed_client_owner),
    )
    assert r.status_code == 201, r.text
    url = r.json()["data"]["inviteLinkUrl"]
    assert url.startswith("http://") or url.startswith("https://")
    assert "/auth/accept-invite#token=" in url
    # Token fragment is at least 32 url-safe chars (secrets.token_urlsafe(32) → 43 chars).
    token_part = url.split("#token=", 1)[1]
    assert len(token_part) >= 32, f"raw token suspiciously short: {len(token_part)} chars"


async def test_revoke_invitation_flips_consumed_and_emits_audit(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
) -> None:
    """D-43-19 — POST /invitations/{id}/revoke atomic-consumes + emits user_invitation_revoked."""
    # Create an invitation.
    r_create = await authed_client_owner.post(
        "/api/v1/users",
        json={"email": "to-revoke@example.com", "fullName": "Revoke Target", "role": "reception"},
        headers=_csrf(authed_client_owner),
    )
    user_id = UUID(r_create.json()["data"]["id"])

    # Look up the invitation token row id.
    token_row = await db_session.scalar(
        select(PasswordResetToken).where(
            PasswordResetToken.user_id == user_id,
            PasswordResetToken.purpose == "invitation",
            PasswordResetToken.consumed_at.is_(None),
        )
    )
    assert token_row is not None
    token_id = token_row.id

    # Revoke via the API.
    r_revoke = await authed_client_owner.post(
        f"/api/v1/users/invitations/{token_id}/revoke",
        json={"reason": "wrong email address"},
        headers=_csrf(authed_client_owner),
    )
    assert r_revoke.status_code == 204, r_revoke.text

    # consumed_at is populated in the DB.
    await db_session.refresh(token_row)
    assert token_row.consumed_at is not None

    # Audit row exists.
    audit_row = (await db_session.execute(
        select(AuditLog).where(
            AuditLog.event == "user_invitation_revoked",
            AuditLog.resource_id == user_id,
        )
    )).scalar_one()
    assert audit_row.payload["invitation_token_id"] == str(token_id)
    assert audit_row.payload["reason"] == "wrong email address"


async def test_revoke_already_consumed_invitation_returns_409(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
) -> None:
    """D-43-19 — revoking an already-consumed invitation returns 409 invitation_already_accepted."""
    r_create = await authed_client_owner.post(
        "/api/v1/users",
        json={"email": "double-revoke@example.com", "fullName": "X", "role": "reception"},
        headers=_csrf(authed_client_owner),
    )
    user_id = UUID(r_create.json()["data"]["id"])
    token_row = await db_session.scalar(
        select(PasswordResetToken).where(
            PasswordResetToken.user_id == user_id, PasswordResetToken.purpose == "invitation",
        )
    )
    token_id = token_row.id

    r1 = await authed_client_owner.post(
        f"/api/v1/users/invitations/{token_id}/revoke",
        json={}, headers=_csrf(authed_client_owner),
    )
    assert r1.status_code == 204

    r2 = await authed_client_owner.post(
        f"/api/v1/users/invitations/{token_id}/revoke",
        json={}, headers=_csrf(authed_client_owner),
    )
    assert r2.status_code == 409, r2.text
    assert r2.json()["detail"]["error"] == "invitation_already_accepted"
