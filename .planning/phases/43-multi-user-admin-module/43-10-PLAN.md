```python
"""Phase 43 USERS-04 + USERS-06 — D-43-20/26/28 end-to-end.

Asserts: deactivate revokes all families → user_deactivated.sessions_revoked_count > 0
→ deactivated user's NEXT /auth/refresh returns 401 with body identical to
invalid_session shape → refresh_failed audit row exists with reason='account_inactive'.
"""
from __future__ import annotations

from uuid import UUID

import pytest
from httpx import AsyncClient
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.audit_models import AuditLog

pytestmark = pytest.mark.asyncio


def _csrf(client: AsyncClient) -> dict[str, str]:
    return {"X-CSRF-Token": client.cookies.get("sportzal_csrf", "")}


async def test_deactivate_revokes_families_and_blocks_refresh(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
    fresh_authed_reception_client: AsyncClient,  # fixture from 43-07b
    fresh_authed_reception_user_id: UUID,        # fixture from 43-07b
) -> None:
    """End-to-end: create + login a reception → owner deactivates → reception refresh fails."""
    # Sanity — reception's refresh works BEFORE deactivation.
    r_before = await fresh_authed_reception_client.post("/api/v1/auth/refresh")
    assert r_before.status_code == 200, (
        f"expected refresh to work before deactivation; got {r_before.status_code} {r_before.text}"
    )

    # Owner deactivates reception.
    r_deact = await authed_client_owner.patch(
        f"/api/v1/users/{fresh_authed_reception_user_id}/deactivate",
        headers=_csrf(authed_client_owner),
    )
    assert r_deact.status_code == 204, r_deact.text

    # Audit row for user_deactivated has sessions_revoked_count > 0 (D-43-28).
    audit_row = (await db_session.execute(
        select(AuditLog).where(
            AuditLog.event == "user_deactivated",
            AuditLog.resource_id == fresh_authed_reception_user_id,
        )
    )).scalar_one()
    assert audit_row.payload["sessions_revoked_count"] >= 1, (
        f"sessions_revoked_count should be >=1 (reception had ≥1 active family); "
        f"got {audit_row.payload['sessions_revoked_count']}"
    )

    # Reception's NEXT refresh attempt returns 401 invalid_session (anti-oracle).
    r_after = await fresh_authed_reception_client.post("/api/v1/auth/refresh")
    assert r_after.status_code == 401, r_after.text
    body = r_after.json()
    # Body shape must match the truly-invalid-session shape.
    assert body.get("detail", {}).get("error") == "invalid_session" or body.get("error") == "invalid_session", (
        f"anti-oracle body shape drift: {body}"
    )

    # Forensic audit row exists (D-43-20).
    refresh_failed_rows = (await db_session.execute(
        select(AuditLog).where(
            AuditLog.event == "refresh_failed",
            AuditLog.resource_type == "session",
        )
    )).scalars().all()
    assert any(
        r.payload.get("user_id") == str(fresh_authed_reception_user_id)
        and r.payload.get("reason") == "account_inactive"
        for r in refresh_failed_rows
    ), (
        f"refresh_failed audit row with reason=account_inactive for user "
        f"{fresh_authed_reception_user_id} missing; rows: {[r.payload for r in refresh_failed_rows]}"
    )


async def test_soft_delete_also_blocks_refresh(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
    fresh_authed_reception_client_2: AsyncClient,
    fresh_authed_reception_user_id_2: UUID,
) -> None:
    """USERS-05 + USERS-06 — soft-delete path also revokes sessions + blocks refresh."""
    r_del = await authed_client_owner.delete(
        f"/api/v1/users/{fresh_authed_reception_user_id_2}",
        headers=_csrf(authed_client_owner),
    )
    assert r_del.status_code == 204, r_del.text

    r_after = await fresh_authed_reception_client_2.post("/api/v1/auth/refresh")
    assert r_after.status_code == 401, r_after.text
```
