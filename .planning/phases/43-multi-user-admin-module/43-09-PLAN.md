```python
"""Phase 43 USERS-04/05/07 guard-condition tests — D-43-16/18/29."""
from __future__ import annotations

from uuid import UUID, uuid4

import pytest
from httpx import AsyncClient
from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.models import User
from app.core.permissions import Role

pytestmark = pytest.mark.asyncio


def _csrf(client: AsyncClient) -> dict[str, str]:
    return {"X-CSRF-Token": client.cookies.get("sportzal_csrf", "")}


async def test_cannot_deactivate_self_returns_409(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
    current_owner_user_id: UUID,  # from 43-07b conftest
) -> None:
    """D-43-16 — owner cannot deactivate own account."""
    r = await authed_client_owner.patch(
        f"/api/v1/users/{current_owner_user_id}/deactivate",
        headers=_csrf(authed_client_owner),
    )
    assert r.status_code == 409, r.text
    assert r.json()["detail"]["error"] == "cannot_deactivate_self"


async def test_cannot_deactivate_last_owner_returns_409(
    db_session: AsyncSession,
    authed_client_owner: AsyncClient,
    single_active_owner_id: UUID,  # from 43-07b conftest
) -> None:
    """D-43-16 — cannot deactivate the last active owner."""
    # The authed_client_owner is a DIFFERENT owner (created via seed); they try to deactivate the last owner.
    # If the only active owner IS the authed client, the test should still pass — guard fires either way.
    r = await authed_client_owner.patch(
        f"/api/v1/users/{single_active_owner_id}/deactivate",
        headers=_csrf(authed_client_owner),
    )
    # Accept both self-guard (if same id) and last-owner-guard outcomes per current_owner_user_id == single_active_owner_id collision.
    assert r.status_code == 409, r.text
    err = r.json()["detail"]["error"]
    assert err in ("cannot_deactivate_last_owner", "cannot_deactivate_self"), err


async def test_cannot_delete_self_returns_409(
    authed_client_owner: AsyncClient,
    current_owner_user_id: UUID,
) -> None:
    """D-43-18 — owner cannot delete own account."""
    r = await authed_client_owner.delete(
        f"/api/v1/users/{current_owner_user_id}",
        headers=_csrf(authed_client_owner),
    )
    assert r.status_code == 409, r.text
    assert r.json()["detail"]["error"] == "cannot_delete_self"


async def test_reception_cannot_create_user_403(
    authed_client_reception: AsyncClient,
) -> None:
    """D-43-29 / USERS-07 — reception is denied any /users mutation (RBAC)."""
    r = await authed_client_reception.post(
        "/api/v1/users",
        json={"email": "x@example.com", "fullName": "X", "role": "reception"},
        headers=_csrf(authed_client_reception),
    )
    assert r.status_code == 403, r.text


async def test_reception_cannot_list_users_403(
    authed_client_reception: AsyncClient,
) -> None:
    """D-43-29 — reception cannot GET /users either (LIST action is OWNER_ONLY)."""
    r = await authed_client_reception.get("/api/v1/users")
    assert r.status_code == 403, r.text


async def test_create_missing_csrf_returns_403(
    authed_client_owner: AsyncClient,
) -> None:
    """D-43-29 — verify_csrf fires on mutation without X-CSRF-Token header."""
    r = await authed_client_owner.post(
        "/api/v1/users",
        json={"email": "no-csrf@example.com", "fullName": "X", "role": "reception"},
        # NO _csrf(...) headers passed
    )
    assert r.status_code == 403, r.text
```
