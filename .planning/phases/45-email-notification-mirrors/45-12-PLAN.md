    Test 2 — AST walker (mirror tests/unit/test_locked_email_templates_ast.py walker pattern):

        def _iter_dispatcher_calls(tree: ast.AST):
            """Yield (call_node, template_id_node) pairs for every dispatcher invocation.

            A dispatcher invocation is a Call whose .func resolves to a get_email_dispatcher()
            chained call OR is named `dispatcher` after assignment from get_email_dispatcher().
            Mirror the walker logic from tests/unit/test_locked_email_templates_ast.py.
            """
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    # Inspect each kwarg for `template_id=...`
                    for kw in node.keywords:
                        if kw.arg == "template_id":
                            yield node, kw.value

        @pytest.mark.parametrize("source_file", PHASE_45_SOURCE_FILES)
        def test_phase45_dispatcher_callsites_are_literal(source_file: Path) -> None:
            full_path = Path(__file__).resolve().parents[2] / source_file
            assert full_path.exists(), f"source not found: {full_path}"
            tree = ast.parse(full_path.read_text(encoding="utf-8"))
            violations = []
            for call_node, tid_node in _iter_dispatcher_calls(tree):
                if not isinstance(tid_node, ast.Constant) or not isinstance(tid_node.value, str):
                    violations.append((source_file, call_node.lineno, ast.dump(tid_node)))
                elif tid_node.value not in LOCKED_EMAIL_TEMPLATES:
                    violations.append((source_file, call_node.lineno, f"unlocked: {tid_node.value!r}"))
            assert not violations, f"non-literal or unlocked template_id callsites: {violations}"
