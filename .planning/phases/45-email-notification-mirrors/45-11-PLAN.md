        @pytest.mark.asyncio
        async def test_payment_receipt_concurrent_fanout_race(
            session_factory_real_commit,  # or whatever the project's name is for real-commit factory
            payment_seed_real_commit,     # fixture that returns a committed payment row, payment_id
        ) -> None:
            payment_id = payment_seed_real_commit.id
            corr_a = uuid4()
            corr_b = uuid4()

            async def insert_receipt(corr: UUID) -> None:
                async with session_factory_real_commit() as session:
                    session.add(PaymentReceipt(
                        payment_id=payment_id,
                        channel='email',
                        audit_correlation_id=corr,
                        to_address='test@example.com',
                    ))
                    await session.commit()

            results = await asyncio.gather(
                insert_receipt(corr_a),
                insert_receipt(corr_b),
                return_exceptions=True,
            )

            integrity_errors = [r for r in results if isinstance(r, IntegrityError)]
            successes = [r for r in results if r is None]
            assert len(integrity_errors) == 1, f"expected 1 IntegrityError, got {results}"
            assert len(successes) == 1, f"expected 1 success, got {results}"

            # Assert exactly one row in DB
            async with session_factory_real_commit() as verify_session:
                from sqlalchemy import select, func
                stmt = select(func.count()).select_from(PaymentReceipt).where(
                    PaymentReceipt.payment_id == payment_id,
                    PaymentReceipt.channel == 'email',
                )
                count = (await verify_session.execute(stmt)).scalar_one()
                assert count == 1

                row_stmt = select(PaymentReceipt.audit_correlation_id).where(
                    PaymentReceipt.payment_id == payment_id,
                    PaymentReceipt.channel == 'email',
                )
                surviving_corr = (await verify_session.execute(row_stmt)).scalar_one()
                assert surviving_corr in (corr_a, corr_b)
