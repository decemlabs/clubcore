    import { screen, waitFor } from '@testing-library/react'

    const navigateMock = vi.fn()
    const useSearchMock = vi.fn(() => ({ page: 1, pageSize: 20, expiring: false }))

    vi.mock('@tanstack/react-router', async () => {
      const actual = await vi.importActual&lt;typeof import('@tanstack/react-router')&gt;('@tanstack/react-router')
      return {
        ...actual,
        useNavigate: () => navigateMock,
        useSearch: () => useSearchMock(),
      }
    })

    vi.mock('@/shared/api/services', () => ({
      services: { memberships: { list: vi.fn() } },
    }))
    import { services } from '@/shared/api/services'
    import { MembershipsListPage } from './MembershipsListPage'

    describe('MembershipsListPage within selector', () => {
      beforeEach(() => {
        vi.clearAllMocks()
        navigateMock.mockReset()
        ;(services.memberships.list as ReturnType&lt;typeof vi.fn&gt;).mockResolvedValue({ items: [], total: 0, page: 1, pageSize: 20 })
      })

      it('does NOT render the within selector when expiring=false', async () => {
        useSearchMock.mockReturnValue({ page: 1, pageSize: 20, expiring: false })
        renderWithProviders(&lt;MembershipsListPage /&gt;, { role: 'owner' })
        await waitFor(() => expect(screen.queryByText('Истекает в течение')).toBeNull())
      })

      it('renders the within selector with default 7 дн. when expiring=true and within is unset', async () => {
        useSearchMock.mockReturnValue({ page: 1, pageSize: 20, expiring: true })

      })

        useSearchMock.mockReturnValue({ page: 3, pageSize: 20, expiring: true, within: 7 })
        renderWithProviders(&lt;MembershipsListPage /&gt;, { role: 'owner' })
        await waitFor(() => expect(screen.getByText('Истекает в течение')).toBeInTheDocument())
        // Open the Select trigger

        const lastCall = navigateMock.mock.calls.at(-1)?.[0]
        const searchFn = lastCall?.search
        const next = typeof searchFn === 'function' ? searchFn({ page: 3, pageSize: 20, expiring: true, within: 7 }) : searchFn
        expect(next.within).toBe(14)
        expect(next.page).toBe(1)
      })

        useSearchMock.mockReturnValue({ page: 1, pageSize: 20, expiring: true })

        await waitFor(() => expect(screen.getByText('Истекает в течение')).toBeInTheDocument())
        const trigger = screen.getByRole('combobox')
        await userEvent.click(trigger)
        for (const n of [1, 3, 7, 14, 30]) {
          const re = new RegExp(`${n} дн\\.`)
          expect(await screen.findByText(re)).toBeInTheDocument()
        }
      })
    })
