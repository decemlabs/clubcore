**`tests/integration/visits/test_visits_concurrent.py` (VIS-TEST-01)** — the headline. Use `db_session_real_commit` fixture:
```python
import asyncio
import pytest
from httpx import AsyncClient

@pytest.mark.asyncio
async def test_concurrent_check_in_one_wins(
    db_session_real_commit,
    authed_client_reception: AsyncClient,
    csrf_headers,
    make_visit_setup,
) -> None:
    """VIS-TEST-01 — 10 parallel POST → exactly 1×201 + 9×409 duplicate_checkin.

    Proves Postgres UNIQUE INDEX uq_visits_client_id_gym_date wins the race
    (D-08), NOT app-layer logic. Uses db_session_real_commit (D-13 sibling)
    because SAVEPOINT-isolated db_session interferes with concurrent insert
    serialisation.
    """
    client_obj, _ = await make_visit_setup()
    headers = csrf_headers(authed_client_reception)
    payload = {"clientId": str(client_obj.id)}

    async def _post():
        return await authed_client_reception.post(
            "/api/v1/visits", json=payload, headers=headers,
        )

    responses = await asyncio.gather(*[_post() for _ in range(10)])
    statuses = sorted(r.status_code for r in responses)
    assert statuses == [201] + [409] * 9, f"Unexpected mix: {statuses}"

    bodies_409 = [r.json() for r in responses if r.status_code == 409]
    assert all(b.get("code") == "duplicate_checkin" for b in bodies_409)

    # Audit row count check
    from sqlalchemy import select, func
    from app.modules.audit.models import AuditLog  # adjust import path
    created_count = await db_session_real_commit.scalar(
        select(func.count()).select_from(AuditLog)
        .where(AuditLog.action == "visit_created")
    )
    rejected_count = await db_session_real_commit.scalar(
        select(func.count()).select_from(AuditLog)
        .where(AuditLog.action == "visit_rejected_duplicate")
    )
    assert created_count == 1
    assert rejected_count == 9
```

async def test_gym_date_generated_column_msk_shifted(
