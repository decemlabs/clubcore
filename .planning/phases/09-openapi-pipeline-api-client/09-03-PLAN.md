        name: Backend (drift gates + statics)
        runs-on: ubuntu-latest
        defaults:
          run:
            working-directory: apps/backend
        steps:
          - name: Checkout
            uses: actions/checkout@v4

          - name: Set up uv (Python 3.12)

          - name: ruff format --check
            run: uv run ruff format --check

          - name: mypy
            run: uv run mypy

            run: uv run lint-imports

        name: Frontend (drift gates + statics)
        runs-on: ubuntu-latest

              version: 9

          - name: Set up Node 20
            uses: actions/setup-node@v4
            with:
              node-version: "20"
              cache: "pnpm"

          - name: pnpm install
            run: pnpm install --frozen-lockfile

          - name: Lint (workspaces)
            run: pnpm -r lint

          - name: Typecheck (workspaces)
            run: pnpm -r typecheck

          - name: Test (workspaces)
            run: pnpm -r test

          - name: Codegen — packages/api-client

          # D-02 / API-07: any backend-schema change that did not regenerate
          # packages/api-client/src/schema.d.ts (locally + commit) fails here.
          - name: Drift gate — packages/api-client/src/schema.d.ts
