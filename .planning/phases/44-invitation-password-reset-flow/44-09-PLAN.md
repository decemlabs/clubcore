    ```python
    def test_password_reset_service_real_callsite_uses_literal_template_id() -> None:
        """D-44-36 / D-41-11 — AST positive-assertion for PASSWORD_RESET_EMAIL.

        Mirrors Phase 43 plan 13 USER_INVITATION_EMAIL pattern. Walker logic
        is inlined here so a future refactor of any shared _iter_dispatcher_calls
        helper cannot silently weaken this gate.
        """
        import ast
        from pathlib import Path

        path = Path(__file__).parent.parent.parent / "app" / "modules" / "auth" / "password_reset_service.py"
        tree = ast.parse(path.read_text())

        callsites = []
        for node in ast.walk(tree):
            # Match calls of the shape get_email_dispatcher()(...)
            if not isinstance(node, ast.Call):
                continue
            inner = node.func
            if not isinstance(inner, ast.Call):
                continue
            inner_name = getattr(inner.func, "id", None) or getattr(inner.func, "attr", None)
            if inner_name != "get_email_dispatcher":
                continue
            callsites.append(node)

        assert len(callsites) == 1, f"expected 1 dispatcher callsite in password_reset_service.py, found {len(callsites)}"

        # The single callsite must pass template_id="PASSWORD_RESET_EMAIL" as a literal string.
        call = callsites[0]
        template_id_kw = next((kw for kw in call.keywords if kw.arg == "template_id"), None)
        assert template_id_kw is not None, "template_id kwarg missing at dispatcher callsite"
        assert isinstance(template_id_kw.value, ast.Constant), (
            f"template_id must be a literal Constant, got {type(template_id_kw.value).__name__}"
        )
        assert template_id_kw.value.value == "PASSWORD_RESET_EMAIL", (
            f"expected template_id='PASSWORD_RESET_EMAIL', got {template_id_kw.value.value!r}"
        )
    ```
