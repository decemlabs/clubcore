    ```python
    """ARQ scheduled job: cleanup expired password_reset_tokens (D-44-31 / D-41-06).

    Daily at 03:30 Europe/Moscow (= 00:30 UTC, container TZ=UTC — see workers
    __init__.py for the offset rationale). 30-day retention per D-44-32:
    rows where `expires_at < NOW() - INTERVAL '30 days'` are DELETEd.

    No audit emit per D-44-31 — housekeeping, not an observable state change.
    Returns the deleted-row count; structlog INFO summary event at the end
    (Phase 18 / Phase 27 convention).

    Transaction ownership: opens its own session via `ctx["sessionmaker"]`
    (Phase 27 D-27-07 pattern b — multi-session). The DELETE + commit is a
    single round-trip; no I/O elsewhere in the body, so a single session
    suffices.
    """

    from __future__ import annotations

    from datetime import timedelta
    from typing import Any

    import structlog
    from sqlalchemy import delete, func

    from app.modules.auth.password_reset_token_model import PasswordResetToken

    _log = structlog.get_logger("workers.scheduled.cleanup_password_reset_tokens")

    _RETENTION = timedelta(days=30)


    async def cleanup_password_reset_tokens(ctx: dict[str, Any]) -> int:
        """Delete password_reset_tokens rows older than the 30-day retention window.

        Returns the deleted-row count; logged via structlog INFO at job end.
        """
        session_factory = ctx["sessionmaker"]
        async with session_factory() as session:
            stmt = delete(PasswordResetToken).where(
                PasswordResetToken.expires_at < func.now() - _RETENTION
            )
            result = await session.execute(stmt)
            await session.commit()
            deleted_count = result.rowcount or 0

        _log.info("cleanup_password_reset_tokens_complete", count=deleted_count)
        return deleted_count
    ```

    Notes:
      - `func.now() - _RETENTION` — SQLAlchemy handles `timedelta` subtraction on `func.now()` via the underlying dialect (Postgres accepts `now() - interval '30 days'`). If type errors at mypy strict, fall back to `text("NOW() - INTERVAL '30 days'")` with proper escaping. Verify by trying the simpler form first.
      - `result.rowcount` is `int | None` in SQLAlchemy 2.0; the `or 0` makes the return type `int` (mypy strict needs the cast).
      - `await session.commit()` is required even for a DELETE (no implicit commit in async SQLAlchemy).
      - No audit emit, no Redis interaction, no email enqueue — pure SQL.
      - The locked summary-event name `cleanup_password_reset_tokens_complete count=N` matches Phase 18 / Phase 27 convention (`<job_name>_complete count=N`).
