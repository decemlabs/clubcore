    ```typescript
    function appendQuery(
      url: string,
      query?: Record<string, string | number | boolean>,
    ): string {
      if (!query) return url
      const sp = new URLSearchParams()
      for (const [k, v] of Object.entries(query)) {
        sp.set(k, String(v))
      }
      const qs = sp.toString()
      if (!qs) return url
      const sep = url.includes('?') ? '&' : '?'
      return `${url}${sep}${qs}`
    }
    ```
