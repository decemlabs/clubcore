```python
"""Phase 51 E2E test — operator-initiated refund through full atomic UoW.

Composition tested:
- 51-08 POST /memberships/{id}/refund endpoint
- 51-07 handle_refund_succeeded webhook handler
- 51-06 _post_commit_enqueue with fiscal_receipt_id branch
- 51-05 dispatch_fiscal_receipt ARQ task (for the kind='refund' receipt)
- 51-03 YooKassaClient.create_receipt + create_refund
"""

@pytest.mark.asyncio
async def test_e2e_refund_full_cycle_membership(
    auth_client_owner,
    seeded_membership_with_succeeded_online_payment,
    yookassa_create_refund_success,        # respx fixture (Phase 48)
    yookassa_get_refund_succeeded,         # respx fixture (plan 51-09 or 51-07 added)
    yookassa_create_receipt_ok,            # respx fixture (plan 51-03)
    worker_runner,                          # ARQ worker test fixture
    db_session,
):
    client, op, original_payment, membership = seeded_membership_with_succeeded_online_payment
    idempotency_key = uuid4()

    # 1. Operator initiates refund via POST endpoint
    resp = await auth_client_owner.post(
        f"/api/v1/online-payments/memberships/{membership.id}/refund",
        json={"idempotency_key": str(idempotency_key), "reason": "customer requested"},
    )
    assert resp.status_code == 202
    body = resp.json()
    online_refund_id = UUID(body["data"]["online_refund_id"])
    yookassa_refund_id = body["data"]["yookassa_refund_id"]
    assert body["data"]["status"] == "pending"

    # Assertions on the DB state after the POST
    refund_row = await db_session.get(OnlineRefund, online_refund_id)
    assert refund_row.status == "pending"
    assert refund_row.yookassa_refund_id == yookassa_refund_id

    # 2. Simulate ЮKassa webhook delivery
    webhook_resp = await auth_client_owner.post(
        "/api/v1/_internal/yookassa/webhook",
        json={
            "event": "refund.succeeded",
            "object": {"id": yookassa_refund_id, "status": "succeeded"},
        },
        # IP-allowlist bypass via the test-mode YOOKASSA_SANDBOX setting or X-Forwarded-For mock
    )
    assert webhook_resp.status_code == 200

    # 3. Assertions on the atomic UoW results (SC#5)
    await db_session.refresh(refund_row)
    assert refund_row.status == "succeeded"
    assert refund_row.succeeded_at is not None

    # 3a. Signed-negative payments row
    refund_payment_row = (await db_session.execute(
        select(Payment).where(Payment.refund_of == original_payment.id)
    )).scalar_one()
    assert refund_payment_row.amount_kopecks == -abs(original_payment.amount_kopecks)
    assert refund_payment_row.method == "online"

    # 3b. Membership transitioned to cancelled + refunded sentinel
    await db_session.refresh(membership)
    assert membership.status == "cancelled"
    assert membership.cancellation_reason == CANCELLATION_REASON_REFUNDED

    # 3c. fiscal_receipts(kind='refund', status='sent') INSERTed
    refund_fr = (await db_session.execute(
        select(FiscalReceipt).where(
            FiscalReceipt.payment_id == refund_payment_row.id,
            FiscalReceipt.kind == "refund",
        )
    )).scalar_one()
    assert refund_fr.status == "sent"

    # 3d. Audit chain: 4 rows with shared audit_correlation_id; root is yookassa_webhook_received
    audit_rows = await db_session.execute(
        select(AuditEvent).where(
            AuditEvent.audit_correlation_id == refund_fr.audit_correlation_id
        ).order_by(AuditEvent.created_at.asc())
    )
    audit_rows = audit_rows.scalars().all()
    audit_events = [r.event for r in audit_rows]
    assert "online_payment_refunded" in audit_events
    assert "membership_refunded" in audit_events
    assert "yookassa_webhook_received" in audit_events
    # chain root last per D-50-18 step 8
    assert audit_rows[-1].event == "yookassa_webhook_received"

    # 4. Worker should now process the kind='refund' fiscal_receipts row
    await worker_runner.drain()  # dequeue + run the dispatch task

    # 5. After dispatch, fr.yookassa_receipt_id should be set
    await db_session.refresh(refund_fr)
    assert refund_fr.yookassa_receipt_id is not None
    # status stays 'sent' — flips to 'succeeded' only via receipt.succeeded webhook
    assert refund_fr.status == "sent"
```

"""Phase 51 E2E test — fiscal receipt lifecycle from payment to settled.

Composition tested:
- Phase 50 handle_payment_succeeded (inserts fiscal_receipts row + post-commit enqueue)
- 51-06 _post_commit_enqueue fiscal-dispatch branch
- 51-05 dispatch_fiscal_receipt ARQ task
- 51-07 handle_receipt_succeeded webhook handler
- 51-03 create_receipt + settings.tax_system_code consumption (FISCAL-07)
"""

async def test_e2e_fiscal_receipt_full_cycle_payment(
    seeded_online_payment_pending,        # client + OnlinePayment(pending)
    yookassa_get_payment_succeeded,        # respx (Phase 48)
    yookassa_create_receipt_ok,            # respx (plan 51-03)
    worker_runner,
    db_session,
    auth_client_owner,
):
    op = seeded_online_payment_pending

    # 1. Simulate payment.succeeded webhook (Phase 50 atomic UoW)
    webhook_resp = await auth_client_owner.post(
        "/api/v1/_internal/yookassa/webhook",
        json={"event": "payment.succeeded", "object": {"id": op.yookassa_payment_id, "status": "succeeded"}},
    )
    assert webhook_resp.status_code == 200

    # Phase 50 ATOMIC UoW result: fiscal_receipts(status='sent') row inserted
    fr = (await db_session.execute(
        select(FiscalReceipt).where(FiscalReceipt.kind == "payment").order_by(FiscalReceipt.created_at.desc()).limit(1)
    )).scalar_one()
    assert fr.status == "sent"
    assert fr.yookassa_receipt_id is None  # not yet dispatched

    # 2. Run the ARQ worker — dispatch_fiscal_receipt should fire
    await worker_runner.drain()
    await db_session.refresh(fr)
    assert fr.yookassa_receipt_id is not None
    assert fr.status == "sent"  # still 'sent' — webhook does the flip

    # 3. Audit row fiscal_receipt_dispatched should exist
    audit_rows = (await db_session.execute(
        select(AuditEvent).where(AuditEvent.event == "fiscal_receipt_dispatched")
    )).scalars().all()
    assert any(r.resource_id == fr.id for r in audit_rows)

    # 4. Simulate receipt.succeeded webhook
    webhook_resp = await auth_client_owner.post(
        "/api/v1/_internal/yookassa/webhook",
        json={"event": "receipt.succeeded", "object": {"id": fr.yookassa_receipt_id, "status": "succeeded"}},
    )
    assert webhook_resp.status_code == 200

    # 5. fr.status flipped to 'succeeded' (FISCAL-04 SC#1)
    await db_session.refresh(fr)
    assert fr.status == "succeeded"
    assert fr.succeeded_at is not None

    # 6. Verify FISCAL-07: the respx-recorded create_receipt request body must have used settings.tax_system_code
    # Implementer: inspect the captured respx call to assert the tax_system_code value matches YooKassaSettings().tax_system_code
