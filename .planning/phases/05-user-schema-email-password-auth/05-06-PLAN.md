    ```
    """Seed the bootstrap owner (AUTH-EP-04 / D-25).

    Idempotent: INSERT ... ON CONFLICT (email) DO NOTHING. Re-running the script
    after the owner exists is a no-op. The compose stack does NOT auto-run this —
    it is a one-shot operator command:

        uv run python -m scripts.seed_demo_data

    Reads SEED_OWNER_EMAIL + SEED_OWNER_PASSWORD from the environment. Both must
    be set; otherwise the script exits 1 with a clear message.

    The created owner has full_name='Owner' (D-01: single column). Operators can
    update it via a future admin endpoint or `psql` once the owner has logged in.
    """

    from __future__ import annotations

    import asyncio
    import os
    import sys

    from sqlalchemy.dialects.postgresql import insert as pg_insert
    from sqlalchemy.ext.asyncio import async_sessionmaker, create_async_engine

    from app.core.config import get_settings
    from app.core.permissions import Role
    from app.core.security import hash_password
    from app.modules.auth.models import User


    async def _run() -> int:
        email = os.environ.get("SEED_OWNER_EMAIL")
        password = os.environ.get("SEED_OWNER_PASSWORD")
        if not email or not password:
            print(
                "SEED_OWNER_EMAIL and SEED_OWNER_PASSWORD must be set "
                "(see .env.example).",
                file=sys.stderr,
            )
            return 1
        if len(password) < 12:
            print(
                "SEED_OWNER_PASSWORD must be at least 12 characters "
                "(AUTH-EP-05 / NIST 800-63B 2024).",
                file=sys.stderr,
            )
            return 1

        email_lower = email.lower()
        settings = get_settings()
        engine = create_async_engine(str(settings.database_url))
        sessionmaker = async_sessionmaker(engine, expire_on_commit=False)
        try:
            async with sessionmaker() as session:
                stmt = (
                    pg_insert(User)
                    .values(
                        email=email_lower,
                        password_hash=await hash_password(password),
                        role=Role.OWNER.value,
                        full_name="Owner",
                    )
                    .on_conflict_do_nothing(index_elements=["email"])
                )
                await session.execute(stmt)
                await session.commit()
                print(f"Seeded owner {email_lower} (idempotent: no-op if existed).")
        finally:
            await engine.dispose()
        return 0


    def main() -> int:
        return asyncio.run(_run())


    if __name__ == "__main__":
        raise SystemExit(main())
    ```

    Notes:
    - `pg_insert` (PostgreSQL dialect-specific) is required for `on_conflict_do_nothing`. The generic `sqlalchemy.insert` doesn't expose UPSERT.
    - `index_elements=["email"]` — Phase 5 plan 03 makes `email` UNIQUE on `users`, so this targets the correct constraint. The auto-generated constraint name is `uq_users_email` per NAMING_CONVENTION.
    - `hash_password` is async; awaiting it inside the values dict is OK because Python evaluates dict values eagerly.
    - The script validates `len(password) >= 12` locally too — defense in depth (the column accepts any text; we don't want a 4-char seed to slip through).

|-----------|----------|-----------|-------------|-----------------|
| T-05.06-01 | Tampering | combined_lifespan ordering | mitigate | db_lifespan enters before redis_lifespan; if Redis fails to start, db engine is already disposed via the async-with unwind |
| T-05.06-02 | Spoofing | register_user_loader race | mitigate | Loader registered BEFORE include_router so the first request sees the bound loader; defensive `InvalidAccessToken('user_loader_not_registered')` in get_current_user covers the cold-path edge |
