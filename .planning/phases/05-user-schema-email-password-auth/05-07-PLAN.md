    @pytest_asyncio.fixture
    async def db_session(app: FastAPI) -> AsyncIterator[AsyncSession]:
        """SAVEPOINT-based per-test rollback against a real Postgres (D-22, TEST-01).

        Pattern (SQLAlchemy 2.0 cookbook "Joining a session into an external
        transaction"):
          1. Open a connection from the engine.
          2. BEGIN an outer transaction on that connection.
          3. Bind an AsyncSession with join_transaction_mode='create_savepoint'
             so service-level session.commit() calls become nested SAVEPOINTs.
          4. Yield the session to the test.
          5. ROLLBACK the outer transaction on teardown — wipes every write
             the test made, even ones a service body claimed to commit.

        Skips cleanly if compose Postgres is unreachable (Phase 3 conftest idiom
        preserved).
        """
        engine = app.state.engine

        async with engine.connect() as connection:
            try:
                await connection.execute(text("select 1"))
            except Exception as exc:  # noqa: BLE001
                pytest.skip(
                    f"DATABASE_URL not reachable; run "
                    f"`docker compose up postgres` first ({exc!r})"
                )

            trans = await connection.begin()
            session_local = async_sessionmaker(
                bind=connection,
                expire_on_commit=False,
                class_=AsyncSession,
                join_transaction_mode="create_savepoint",
            )
            async with session_local() as session:
                try:
                    yield session
                finally:
                    await trans.rollback()
    ```

    Notes:
    - `engine.connect()` returns an `AsyncConnection`; the `async with` ensures the connection is returned to the pool.
    - `connection.begin()` returns an `AsyncTransaction` — `await trans.rollback()` is the correct teardown.
    - `join_transaction_mode="create_savepoint"` is the new (SQLAlchemy 2.0+) parameter name; do NOT use the old `nested=True` flag.
    - Service code in the auth service (Plan 04) uses `async with session.begin():` blocks. Under SAVEPOINT mode, those become nested savepoints under the outer transaction — they release/rollback within the outer scope without committing anything externally.
    - The `text("select 1")` connectivity probe is preserved so missing-Postgres tests skip rather than ERROR.

    """Smoke test for the SAVEPOINT-based db_session fixture (TEST-01 / D-22).

    Asserts that a service-style `session.commit()` inside a test becomes a nested
    SAVEPOINT under the outer transaction, and that the outer rollback wipes the
    write across test runs. If two consecutive tests insert the same UNIQUE email
    without raising IntegrityError, the SAVEPOINT pattern is working.
    """

    from __future__ import annotations

    from sqlalchemy import select
    from sqlalchemy.ext.asyncio import AsyncSession

    from app.core.permissions import Role
    from app.core.security import hash_password
    from app.modules.auth.models import User


    async def _insert_owner(db_session: AsyncSession, email: str) -> User:
        user = User(
            email=email,
            password_hash=await hash_password("hunter22hunter22"),
            role=Role.OWNER,
            full_name="Owner",
        )
        db_session.add(user)
        await db_session.commit()  # service-style commit — should become nested SAVEPOINT
        return user


    async def test_savepoint_first_insert(db_session: AsyncSession) -> None:
        """Inserts an owner; relies on the next test to prove rollback happened."""
        await _insert_owner(db_session, "savepoint-smoke@test.local")
        found = await db_session.scalar(
            select(User).where(User.email == "savepoint-smoke@test.local")
        )
        assert found is not None
        assert found.email == "savepoint-smoke@test.local"


    async def test_savepoint_second_insert_same_email(db_session: AsyncSession) -> None:
        """Same email as previous test — must succeed, proving the previous commit was rolled back."""
        await _insert_owner(db_session, "savepoint-smoke@test.local")
        found = await db_session.scalar(
            select(User).where(User.email == "savepoint-smoke@test.local")
        )
        assert found is not None
    ```

    Notes:
    - The two tests intentionally use the SAME email. With the OLD `session.rollback()` fixture this pair would fail on `IntegrityError(uq_users_email)` because the first test's commit would survive. With SAVEPOINT-based rollback, the first test's commit becomes a nested savepoint under the outer transaction, and the outer trans.rollback() wipes it on teardown.
    - Run order matters: pytest collects in alphabetical/declaration order by default. If test ordering becomes an issue, both tests stand independently but the contract is asserted across the pair.
    - These tests need a running Postgres + applied migrations. The fixture skips if Postgres is unreachable (preserves CI behavior).
