```
@pytest.mark.asyncio
async def test_otp_email_anti_oracle(
    async_client: AsyncClient,
    four_otp_fixture_users: dict[str, str],
) -> None:
    """AUTH-EM-04 (D-42-24) -- 4-case anti-oracle gate.

    Cases A, C, D send {channel:'email', email:<x>} and MUST produce
    byte-identical 202 + bounded-equal timing within 100ms (RESET-06
    lineage). Case B sends no `channel` field -- exercises Telegram-default
    backwards-compat path and is asserted separately for status 202 only.
    """
    emails_email_channel = [
        ("case_a", four_otp_fixture_users["case_a"]),
        ("case_c", four_otp_fixture_users["case_c"]),
        ("case_d", four_otp_fixture_users["case_d"]),
    ]
    responses: list[tuple[str, int, bytes, float]] = []
    for name, email in emails_email_channel:
        t0 = time.perf_counter()
        resp = await async_client.post(
            "/api/v1/auth/otp/request",
            json={"channel": "email", "email": email},
        )
        t1 = time.perf_counter()
        responses.append((name, resp.status_code, resp.content, t1 - t0))

    # Status-code parity
    statuses = {r[1] for r in responses}
    assert statuses == {202}, f"non-uniform statuses (anti-oracle leak): {responses}"

    # Body parity -- byte-for-byte
    bodies = {r[2] for r in responses}
    assert len(bodies) == 1, (
        "response body diverges across 3 email-channel cases -- anti-oracle leak: "
        f"{[(r[0], r[2]) for r in responses]}"
    )

    # Timing parity -- bounded-equal within 100ms (D-42-24 + RESET-06 precedent)
    timings = [r[3] for r in responses]
    assert max(timings) - min(timings) < 0.100, (
        "timing oracle on email-channel: "
        f"max-min={max(timings) - min(timings):.3f}s > 100ms; "
        f"per-case timings={list(zip([r[0] for r in responses], timings))}"
    )

    # Case B -- Telegram-default backwards-compat -- separate assertion
    resp_b = await async_client.post(
        "/api/v1/auth/otp/request",
        json={"email": four_otp_fixture_users["case_b"]},
    )
    assert resp_b.status_code == 202, resp_b.status_code
```

```
def test_email_otp_login_real_callsite_present() -> None:
    """Phase 42 D-42-27 -- assert the EMAIL_OTP_LOGIN callsite in
    app/modules/auth/service.py is discovered by the AST walker and that the
    template_id is a literal string constant (not an expression).

    Phase 41 INFRA-36 AST gate's first real-code exercise -- protects against
    later 'helpfully refactored' const-extraction silently bypassing the gate.
    """
    from pathlib import Path
    import ast

    service_py = Path(__file__).resolve().parents[2] / "app" / "modules" / "auth" / "service.py"
    tree = ast.parse(service_py.read_text(encoding="utf-8"))

    found_literal_template_ids: list[str] = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue
        for kw in node.keywords:
            if kw.arg == "template_id" and isinstance(kw.value, ast.Constant) and isinstance(kw.value.value, str):
                found_literal_template_ids.append(kw.value.value)

    assert "EMAIL_OTP_LOGIN" in found_literal_template_ids, (
        f"Phase 42 D-42-27 violation: EMAIL_OTP_LOGIN literal callsite "
        f"not found in app/modules/auth/service.py. Got: {found_literal_template_ids}"
    )
```
