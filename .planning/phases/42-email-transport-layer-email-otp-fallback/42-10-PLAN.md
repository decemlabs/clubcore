Handler:

```
@router.post(
    "/webhook",
    status_code=202,
    response_class=Response,
)
async def email_webhook(
    request: Request,
    session: Annotated[AsyncSession, Depends(get_db)],
) -> Response:
    """Yandex Cloud Postbox bounce/complaint webhook (D-42-17 / EMAIL-07).

    Flow:
      1. Read raw body bytes (no parse yet).
      2. Compute HMAC-SHA256 of raw body with settings.email.webhook_secret.
      3. hmac.compare_digest against X-Email-Webhook-Signature header.
      4. Mismatch / missing -> 401 (no audit emit -- would amplify noise from
         unsigned probes; matches auth.router silent-401 discipline).
      5. Parse body, route by eventType -> UPDATE EmailSendLog status.
      6. On hard-bounce/complaint: audit.emit('email_send_failed', ...) with
         classification reason.
    """
    settings = get_settings()
    raw_body = await request.body()

    presented_sig = request.headers.get("x-email-webhook-signature", "")
    expected_sig = hmac.new(
        settings.email.webhook_secret.get_secret_value().encode("utf-8"),
        raw_body,
        hashlib.sha256,
    ).hexdigest()
    if not presented_sig or not hmac.compare_digest(presented_sig, expected_sig):
        # 401 BEFORE parse. Do NOT emit audit -- this would amplify probe noise.
        raise HTTPException(status_code=401, detail="invalid_webhook_signature")

    # Past the gate -- now safe to parse.
    try:
        payload = json.loads(raw_body)
    except json.JSONDecodeError as exc:
        raise HTTPException(status_code=400, detail="invalid_payload") from exc

    # Yandex Cloud Postbox event shape mirrors AWS SES eventBridge:
    # { "eventType": "Bounce"|"Complaint"|"Delivery"|...,
    #   "mail": { "messageId": "<provider_message_id>", ... },
    #   "bounce": { "bounceType": "Permanent"|"Transient", ... } }
    # Exact shape verified live during Phase 46 VER-12 probe -- if upstream
    # differs, adapt parsing here. The fields cited (eventType + mail.messageId
    # + bounce.bounceType) are the canonical SES-V2 contract that Postbox
    # exposes per D-42-01.
    event_type = payload.get("eventType")
    mail = payload.get("mail") or {}
    provider_message_id = mail.get("messageId")

    if not provider_message_id:
        _log.warning("email_webhook_missing_message_id", event_type=event_type)
        return Response(status_code=202)

    # Resolve target row.
    row = await session.scalar(
        select(EmailSendLog).where(EmailSendLog.provider_message_id == provider_message_id)
    )
    if row is None:
        _log.warning("email_webhook_unknown_message_id", provider_message_id=provider_message_id, event_type=event_type)
        return Response(status_code=202)

    new_status: str | None = None
    new_bounce_type: str | None = None
    emit_reason: str | None = None

    if event_type == "Bounce":
        bounce_type = (payload.get("bounce") or {}).get("bounceType")
        if bounce_type == "Permanent":
            new_status = "bounced"
            new_bounce_type = "hard"
            emit_reason = "bounced_hard"
        else:
            # Transient (soft) bounce -- D-42-19 stays quiet.
            new_status = "bounced"
            new_bounce_type = "soft"
            emit_reason = None
    elif event_type == "Complaint":
        new_status = "complained"
        new_bounce_type = None
        emit_reason = "complained"
    elif event_type == "Delivery":
        new_status = "delivered"
        new_bounce_type = None
        emit_reason = None
    else:
        _log.warning("email_webhook_unknown_event_type", event_type=event_type)
        return Response(status_code=202)

    await session.execute(
        update(EmailSendLog)
        .where(EmailSendLog.id == row.id)
        .values(status=new_status, bounce_type=new_bounce_type)
    )

    if emit_reason is not None:
        # Pitfall 2: emit BEFORE commit.
        await audit.emit(
            session,
            "email_send_failed",
            actor_user_id=None,
            resource_type="email_send_log",
            resource_id=row.id,
            payload=EmailSendFailedPayload(
                audit_correlation_id=row.audit_correlation_id,
                template_id=row.template_id,
                to_email=row.to_address,
                reason=emit_reason,
                provider_error_code=None,
            ),
        )

    await session.commit()
    return Response(status_code=202)
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-42-10-01 | Spoofing | attacker posts forged bounce events to mark sends as bounced | mitigate | HMAC-SHA256 signature in `X-Email-Webhook-Signature` header verified with `hmac.compare_digest` (timing-safe) BEFORE body parse. Secret lives in `EmailProviderSettings.webhook_secret: SecretStr` (plan 42-02) and is provisioned at the Postbox console |
| T-42-10-02 | Tampering | malformed payload with crafted provider_message_id targeting another tenant | mitigate | EmailSendLog SELECT scopes to `WHERE provider_message_id = ?` — within a single-zal deployment there is no tenant isolation requirement; unknown IDs return 202 with structlog WARN |
| T-42-10-03 | Information disclosure | timing oracle on signature verification | mitigate | `hmac.compare_digest` is constant-time. Missing header is checked alongside signature mismatch (single 401 branch) |
| T-42-10-04 | Denial of service | unsigned-probe flood | mitigate | O(1) rejection BEFORE body parse and BEFORE DB access; no audit emit (avoid amplification); FastAPI request-body read is bounded by upstream proxy limits |
| T-42-10-05 | Repudiation | bounce event UPDATE without audit on hard-bounce/complaint | mitigate | `audit.emit('email_send_failed', ...)` runs BEFORE `session.commit()` (Pitfall 2 / PATTERNS.md §A) inside same UoW as the EmailSendLog UPDATE |
| T-42-10-06 | Elevation of privilege | malicious webhook flips email_verified=false to lock out a user | accept | Phase 42 EXPLICITLY does NOT touch `email_verified` from the webhook (D-42-19 — passive logging only; aggressive flag flipping deferred to v1.7). Lockout-from-webhook is impossible by design |
