```python
class ClientCreateRequest(RequestContract):
    last_name: str = Field(min_length=1, max_length=128)
    first_name: str = Field(min_length=1, max_length=128)
    middle_name: str | None = Field(default=None, max_length=128)
    phone: str = Field(pattern=PHONE_REGEX)  # required + E.164 (D-10)
    email: EmailStr | None = None
    birthday: date | None = None
    gender: Gender | None = None
    tags: list[str] = Field(default_factory=list)
    notes: str | None = Field(default=None, max_length=4096)
    emergency_contact: EmergencyContact | None = None
    telegram_user_id: int | None = None  # bigint; service may set, frontend usually doesn't

    @field_validator("tags")
    @classmethod
    def validate_tags(cls, v: list[str]) -> list[str]:
        """D-16: lowercase, max 16 tags, each <=32 chars, regex ^[a-z0-9а-я\\-_]+$."""
        if len(v) > MAX_TAGS:
            raise ValueError(f"Maximum {MAX_TAGS} tags per client")
        result: list[str] = []
        for tag in v:
            normalised = tag.strip().lower()
            if len(normalised) == 0:
                raise ValueError("Empty tag not allowed")
            if len(normalised) > MAX_TAG_LENGTH:
                raise ValueError(f"Tag '{normalised}' exceeds {MAX_TAG_LENGTH} characters")
            if not TAG_REGEX.match(normalised):
                raise ValueError(f"Tag '{normalised}' contains invalid characters")
            result.append(normalised)
        return result
```

```python
class ClientUpdateRequest(RequestContract):
    """PATCH semantics: model_dump(exclude_unset=True). D-01 forbids explicit null —
    use absence of key to leave field unchanged.
    """

    last_name: str | None = Field(default=None, min_length=1, max_length=128)
    first_name: str | None = Field(default=None, min_length=1, max_length=128)
    middle_name: str | None = Field(default=None, max_length=128)
    phone: str | None = Field(default=None, pattern=PHONE_REGEX)
    email: EmailStr | None = None
    birthday: date | None = None
    gender: Gender | None = None
    tags: list[str] | None = None
    notes: str | None = Field(default=None, max_length=4096)
    emergency_contact: EmergencyContact | None = None
    telegram_user_id: int | None = None

    @model_validator(mode="before")
    @classmethod
    def reject_explicit_null(cls, data: Any) -> Any:
        """D-01: null-out via PATCH not supported in v1.1; reject {"key": null} payloads."""
        if isinstance(data, dict):
            null_keys = [k for k, v in data.items() if v is None]
            if null_keys:
                raise ValueError(
                    f"Explicit null not supported for: {sorted(null_keys)}. "
                    "Omit the key to leave the field unchanged."
                )
        return data

    @field_validator("tags")
    @classmethod
    def validate_tags(cls, v: list[str] | None) -> list[str] | None:
        """D-16: same rules as create. None bypasses validation entirely (PATCH-absent)."""
        if v is None:
            return v
        if len(v) > MAX_TAGS:
            raise ValueError(f"Maximum {MAX_TAGS} tags per client")
        result: list[str] = []
        for tag in v:
            normalised = tag.strip().lower()
            if len(normalised) == 0:
                raise ValueError("Empty tag not allowed")
            if len(normalised) > MAX_TAG_LENGTH:
                raise ValueError(f"Tag '{normalised}' exceeds {MAX_TAG_LENGTH} characters")
            if not TAG_REGEX.match(normalised):
                raise ValueError(f"Tag '{normalised}' contains invalid characters")
            result.append(normalised)
        return result
```

```python
from app.core.pagination import PageQuery

class ClientListQuery(PageQuery):
    """List filters per CLIENTS-04 + D-14."""

    q: str | None = Field(default=None, max_length=128)
    tag: str | None = Field(default=None, max_length=MAX_TAG_LENGTH)
    gender: Gender | None = None
    created_from: datetime | None = None  # → createdFrom on wire (D-14 inclusive)
    created_to: datetime | None = None  # → createdTo on wire (D-14 inclusive)
    has_telegram: bool | None = None  # → hasTelegram on wire
    sort: ClientSort = ClientSort.CREATED_AT_DESC

    @field_validator("created_from", "created_to", mode="before")
    @classmethod
    def normalise_date_boundary(cls, v: Any, info: Any) -> Any:
        """D-14: date-only inputs are normalised — created_from to 00:00:00 UTC,
        created_to to 23:59:59.999999 UTC. datetime inputs pass through unchanged.
        """
        if v is None:
            return v
        if isinstance(v, str):
            # Try parsing as date-only YYYY-MM-DD; if it has time component, leave it.
            if len(v) == 10 and v.count("-") == 2:
                from datetime import datetime as _dt, timezone
                parsed = _dt.strptime(v, "%Y-%m-%d").replace(tzinfo=timezone.utc)
                if info.field_name == "created_to":
                    return parsed.replace(hour=23, minute=59, second=59, microsecond=999999)
                return parsed
        if isinstance(v, date) and not isinstance(v, datetime):
            from datetime import datetime as _dt, timezone
            base = _dt(v.year, v.month, v.day, tzinfo=timezone.utc)
            if info.field_name == "created_to":
                return base.replace(hour=23, minute=59, second=59, microsecond=999999)
            return base
        return v

    @field_validator("q")
    @classmethod
    def normalise_q(cls, v: str | None) -> str | None:
        """D-12: q < 2 chars → ignore (return None so service skips filter)."""
        if v is None:
            return None
        stripped = v.strip()
        if len(stripped) < 2:
            return None
        return stripped
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-12 | Tampering | Phone format injection (e.g., `'+7' OR 1=1`) | mitigate | `Field(pattern=PHONE_REGEX)` — Pydantic rejects anything not matching `^\+[1-9]\d{1,14}$`. SQL parameterisation in repository (Plan 04) is the second line. |
| T-08-13 | Tampering | tags array exhaustion (memory DoS) | mitigate | `MAX_TAGS=16`, `MAX_TAG_LENGTH=32`, regex restricts char set. ARRAY size capped at 16; per-tag at 32 chars. Worst case: 512 chars per request. |
| T-08-14 | Tampering | Explicit null PATCH semantics divergence | mitigate | `model_validator(mode="before")` rejects `{"key": null}` per D-01. Frontend cannot ambiguously "clear" a field; UI must omit the key. |
| T-08-15 | DoS | q='' empty / 1-char triggers full-table scan | mitigate | `field_validator("q")` returns None for `len(stripped) < 2` per D-12; service treats None as "no filter applied". |
| T-08-16 | Information Disclosure | ClientResponse leaks deleted_at | mitigate | `ClientResponse` does not declare `deleted_at`; soft-deleted rows are 404'd at the repository boundary (Plan 04 `get_alive`); even if a soft-deleted row leaked, response shape would not expose the deleted_at column. |

| T-08-18 | Spoofing | gender enum bypass with arbitrary value | mitigate | Pydantic `gender: Gender | None` rejects anything outside `{'male', 'female'}`. DB-level CHECK constraint is the second line. |
