   ```python
   def test_rfc2047_cyrillic_subject_roundtrip() -> None:
       # Render the locked Russian template with fixture vars:
       rendered_subject = EMAIL_EXPIRING_7D_VARIANT_A.render_subject(                  # adjust to actual API
           {"client_name": "Тест", "expires_at_human": "20 мая (через 7 дней)"},
       )
       # Sanity: the source string contains Cyrillic glyphs
       assert any("а" <= c <= "я" or "А" <= c <= "Я" for c in rendered_subject)

       envelope = EmailEnvelope(
           to_email="roundtrip+rfc2047@local.dev",
           subject=rendered_subject,
           html_body="<p>round-trip test</p>",
           text_body="round-trip test",
       )

       # Serialise via the same path used in production (the dispatcher's
       # message-building helper). If a `to_mime_message()` or
       # `serialise_headers()` method exists, use it; otherwise call
       # `email.message.EmailMessage()` and set the subject directly to
       # mimic the dispatcher.
       serialised_subject_header = envelope.encoded_subject()                           # adjust to actual method
       # serialised_subject_header is a str like '=?UTF-8?B?0JLQsNGIIA==?=' etc.

       # Decode + assert byte-identical:
       decoded = str(make_header(decode_header(serialised_subject_header)))
       assert decoded == rendered_subject, (
           f"RFC 2047 round-trip failed:\n"
           f"  original (len={len(rendered_subject)}): {rendered_subject!r}\n"
           f"  decoded  (len={len(decoded)}): {decoded!r}"
       )

       # NBSP preservation: if the rendered subject contains  , the decoded MUST too
       if " " in rendered_subject:
           assert " " in decoded, "NBSP lost during round-trip"
   ```
