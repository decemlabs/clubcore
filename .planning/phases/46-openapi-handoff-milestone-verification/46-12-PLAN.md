"""Phase 46 / VER-12 — live email-deliverability probe.

Sends one email each to a yandex.ru + mail.ru + rambler.ru recipient via the

Yandex Postbox credentials in staging or a throwaway prod-equivalent).
Captures the per-send ``provider_message_id`` and prints them. The operator
then manually fetches the ``Authentication-Results:`` header line from each
recipient's mailbox ("show original" / "view source") and pastes it into
``.planning/milestones/v1.6-VERIFICATION-LOG.md`` under the
``email_deliverability_probe:`` YAML block.

NO IMAP automation per D-46-19 — keeps the probe tiny + avoids credential
plumbing for 3 mailbox providers.

Run from `apps/backend/`:

    PROBE_YANDEX_TO=foo+probe@yandex.ru \
    PROBE_MAIL_TO=foo+probe@mail.ru \
    PROBE_RAMBLER_TO=foo+probe@rambler.ru \
    EMAIL_PROVIDER_API_KEY=<real-key> \
    uv run python -m scripts.verify.v1_6_email_probe

Re-uses ``EMAIL_OTP_LOGIN`` template (D-46-21) with a fixture OTP code so
the body is locked Russian copy already on the v1.6 sign-off list — avoids
introducing a probe-only template.
"""

from __future__ import annotations

import asyncio
import os
import sys
import uuid

# Env setdefault — mirrors export_openapi.py:28-46 (must precede app.* imports)
os.environ.setdefault("ENVIRONMENT", "staging")            # probe runs against staging by default
os.environ.setdefault("DATABASE_URL", "postgresql+asyncpg://app:app@localhost:5432/sportzal")
os.environ.setdefault("REDIS_URL", "redis://localhost:6379/0")
os.environ.setdefault("SECRET_KEY", "x" * 48)              # placeholder; probe doesn't sign anything cryptographic
os.environ.setdefault("TELEGRAM_BOT_TOKEN", "placeholder")
os.environ.setdefault("TELEGRAM_BOT_USERNAME", "placeholder")
# EMAIL_PROVIDER + EMAIL_PROVIDER_API_KEY MUST come from operator env (NO setdefault):
if not os.environ.get("EMAIL_PROVIDER_API_KEY"):
    print("FATAL: EMAIL_PROVIDER_API_KEY must be set (no default).", file=sys.stderr)
    print("       Export the staging Yandex Postbox API key before running.", file=sys.stderr)
    sys.exit(2)

RECIPIENTS = {
    "yandex.ru": os.environ.get("PROBE_YANDEX_TO"),
    "mail.ru":   os.environ.get("PROBE_MAIL_TO"),
    "rambler.ru": os.environ.get("PROBE_RAMBLER_TO"),
}
missing = [provider for provider, addr in RECIPIENTS.items() if not addr]
if missing:
    print(f"FATAL: missing env-var recipient(s) for: {missing}", file=sys.stderr)
    print("       Set PROBE_YANDEX_TO + PROBE_MAIL_TO + PROBE_RAMBLER_TO.", file=sys.stderr)
    sys.exit(2)

# Now safe to import the app
from app.integrations.email.dispatcher import get_email_dispatcher    # adjust path per actual export


async def main() -> int:
    dispatcher = get_email_dispatcher()
    fixture_otp = "000000"     # placeholder — probe body is locked OTP-login template; OTP is throwaway

    results: dict[str, str] = {}
    for provider, addr in RECIPIENTS.items():
        try:
            result = await dispatcher.dispatch(
                template_id="EMAIL_OTP_LOGIN",                                          # LITERAL — LOCKED_EMAIL_TEMPLATES AST gate
                to_email=addr,
                template_vars={"otp_code": fixture_otp},
            )
            # The dispatch return shape MAY be a dict, dataclass, or EmailSendLog row.
            # Read dispatcher.py to find the actual return type; adjust this extraction:
            provider_message_id = (
                getattr(result, "provider_message_id", None)
                or (result.get("provider_message_id") if isinstance(result, dict) else None)
                or "<unknown — adjust extraction per dispatcher return shape>"
            )
            results[provider] = provider_message_id
            print(f"{provider}: {provider_message_id}")
        except Exception as exc:  # noqa: BLE001
            print(f"{provider}: FAIL ({exc!r})", file=sys.stderr)
            results[provider] = f"FAIL: {exc!r}"

    # Print a final machine-readable summary the operator can copy into the verification log:
    print()
    print("--- copy into v1.6-VERIFICATION-LOG.md `email_deliverability_probe:` block ---")
    print("email_deliverability_probe:")
    for provider, addr in RECIPIENTS.items():
        print(f"  {provider.replace('.', '_')}:")
        print(f"    to: \"<redact-to-domain-only-per-T-46-12-01>\"  # actual: {addr}")
        print(f"    provider_message_id: \"{results.get(provider, '')}\"")
        print(f"    authentication_results: \"<paste verbatim from recipient mailbox 'show original'>\"")
        print(f"    timestamp: \"\"")

    # Exit 0 only if all three sends got a non-FAIL provider_message_id:
    failed = [p for p, mid in results.items() if mid.startswith("FAIL")]
    return 1 if failed else 0


if __name__ == "__main__":
    raise SystemExit(asyncio.run(main()))
