   ```python
   @pytest.mark.asyncio
   async def test_bounce_webhook_active_send_race(
       real_commit_engine: AsyncEngine,
   ) -> None:
       session_factory = async_sessionmaker(real_commit_engine, expire_on_commit=False)

       recipient = f"race-bounce+{uuid4().hex[:8]}@local.dev"
       existing_send_id = uuid4()
       provider_message_id = f"msg-{uuid4().hex}"

       # ── Seed an existing email_send_log row in 'sent' status ──
       async with session_factory() as setup:
           await setup.execute(text("""
               INSERT INTO email_send_log (id, recipient_email, template_id, status, provider_message_id, created_at)
               VALUES (:id, :rcpt, 'EMAIL_OTP_LOGIN', 'sent', :pmid, now())
           """), {"id": existing_send_id, "rcpt": recipient, "pmid": provider_message_id})
           await setup.commit()

       app = create_app()

       # ── Race: HMAC-signed webhook POST || fresh EmailDispatcher.dispatch ──
       from app.integrations.email.dispatcher import get_email_dispatcher
       dispatcher = get_email_dispatcher()

       # Body shape MUST match the SES-V2 contract the router parses
       # (router.py:118-120): eventType + mail.messageId + bounce.bounceType
       webhook_body_dict = {
           "eventType": "Bounce",
           "mail": {"messageId": provider_message_id},
           "bounce": {"bounceType": "Permanent"},
       }
       webhook_body = json.dumps(webhook_body_dict).encode("utf-8")
       webhook_signature = _sign(webhook_body)

       async def _bounce_webhook() -> int:
           async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as client:
               r = await client.post(
                   "/api/v1/_internal/email/webhook",
                   headers={"x-email-webhook-signature": webhook_signature},
                   content=webhook_body,   # raw bytes — signature is over raw body, not re-serialised JSON
               )
               return r.status_code

       async def _fresh_dispatch() -> None:
           # Use the dispatcher API directly (no router involvement; this is the active-send branch)
           await dispatcher.dispatch(
               template_id="EMAIL_OTP_LOGIN",                                            # LITERAL — required by LOCKED_EMAIL_TEMPLATES AST gate
               to_email=recipient,
               template_vars={"otp_code": "123456"},
           )

       results = await asyncio.gather(_bounce_webhook(), _fresh_dispatch(), return_exceptions=True)

       # ── Verify outcome ──
       async with session_factory() as verify:
           # Existing row → bounced
           existing_status = (await verify.execute(text(
               "SELECT status FROM email_send_log WHERE id = :id"
           ), {"id": existing_send_id})).scalar_one()
           assert existing_status == "bounced", f"existing row expected 'bounced', got {existing_status}"

           # Fresh row → sent (or 'queued' if dispatcher is async; accept both per Phase 42 D-42 lifecycle)
           fresh_count = (await verify.execute(text(
               "SELECT count(*) FROM email_send_log WHERE recipient_email = :rcpt AND id <> :existing AND status IN ('sent', 'queued')"
           ), {"rcpt": recipient, "existing": existing_send_id})).scalar_one()
           assert fresh_count == 1, f"fresh send row expected, got {fresh_count} new rows"
   ```
