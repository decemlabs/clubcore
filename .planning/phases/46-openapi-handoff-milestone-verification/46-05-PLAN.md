   ```python
   @pytest.mark.asyncio
   async def test_password_reset_token_replay_race(
       real_commit_engine: AsyncEngine,
   ) -> None:
       session_factory = async_sessionmaker(real_commit_engine, expire_on_commit=False)

       # ── Seed a user + a fresh unconsumed token ─────────
       raw_token = uuid4().hex + uuid4().hex                                   # 64 hex chars
       user_id = uuid4()

       async with session_factory() as setup:
           setup.add(User(
               id=user_id,
               email=f"race+{user_id}@local.dev",
               full_name="Race Test User",
               role="reception",
               is_active=True,
               password_hash=hash_password("OldPass123!"),
           ))
           await setup.flush()
           # Use the existing service to mint a real token row (avoids hash-shape drift):
           from app.modules.auth.password_reset_service import _hash_token_for_storage  # or the actual helper
           setup.add(PasswordResetToken(
               id=uuid4(),
               user_id=user_id,
               purpose="password_reset",                                       # match repository semantics
               token_hash=_hash_token_for_storage(raw_token),
               expires_at=text("now() + interval '1 hour'"),
           ))
           await setup.commit()

       # ── Race: two concurrent confirms with the same raw token ──
       app = create_app()
       async def _confirm() -> int:
           async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as client:
               resp = await client.post(
                   "/api/v1/auth/password-reset/confirm",
                   json={"token": raw_token, "new_password": "NewPass456!"},
               )
               return resp.status_code

       statuses = await asyncio.gather(_confirm(), _confirm(), return_exceptions=False)

       # ── Assertions ──
       assert sorted(statuses) == [200, 410], f"expected one 200 + one 410, got {sorted(statuses)}"

       async with session_factory() as verify:
           audit_rows = (await verify.execute(text(
               "SELECT count(*) FROM audit_log WHERE action = 'password_reset_completed' AND target_user_id = :uid"
           ), {"uid": user_id})).scalar_one()
           assert audit_rows == 1, f"expected exactly one password_reset_completed audit row, got {audit_rows}"

           consumed_rows = (await verify.execute(text(
               "SELECT count(*) FROM password_reset_tokens WHERE user_id = :uid AND consumed_at IS NOT NULL"
           ), {"uid": user_id})).scalar_one()
           assert consumed_rows == 1
   ```
