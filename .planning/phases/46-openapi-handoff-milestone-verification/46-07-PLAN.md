3. Test body:
   ```python
   @pytest.mark.asyncio
   async def test_deactivate_refresh_race(
       real_commit_engine: AsyncEngine,
   ) -> None:
       session_factory = async_sessionmaker(real_commit_engine, expire_on_commit=False)

       owner_id, target_id = uuid4(), uuid4()
       target_password = "TargetPass123!"

       async with session_factory() as setup:
           setup.add(User(id=owner_id, email=f"owner+{owner_id}@local.dev",
                          full_name="Race Owner", role="owner", is_active=True,
                          password_hash=hash_password("OwnerPass123!")))
           setup.add(User(id=target_id, email=f"target+{target_id}@local.dev",
                          full_name="Race Target", role="reception", is_active=True,
                          password_hash=hash_password(target_password)))
           await setup.commit()

       app = create_app()
       async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as owner_client, \
                  AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as target_client:

           owner_login = await owner_client.post("/api/v1/auth/login",
               json={"email": f"owner+{owner_id}@local.dev", "password": "OwnerPass123!"})
           assert owner_login.status_code == 200

           target_login = await target_client.post("/api/v1/auth/login",
               json={"email": f"target+{target_id}@local.dev", "password": target_password})
           assert target_login.status_code == 200

           # ── Race: owner deactivates || target refreshes ──
           async def _deactivate() -> int:
               r = await owner_client.post(f"/api/v1/users/{target_id}/deactivate")
               return r.status_code

           async def _refresh() -> tuple[int, dict[str, Any]]:
               r = await target_client.post("/api/v1/auth/refresh")
               return r.status_code, (r.json() if r.content else {})

           deact_status, (refresh_status, refresh_body) = await asyncio.gather(_deactivate(), _refresh())

       assert deact_status in {200, 204}, f"deactivate expected 200/204, got {deact_status}"
       # Refresh outcome depends on race order:
       # - deactivate-first → 401 account_inactive (Branch A)
       # - refresh-first then deactivate → 200 (the refresh slipped in BEFORE the deactivate UoW)
       # If refresh slipped in, the NEXT call MUST fail. Issue a second refresh attempt:
       if refresh_status == 200:
           async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as recheck:
               # Replay the refresh cookies from target_client
               recheck.cookies = target_client.cookies
               second = await recheck.post("/api/v1/auth/refresh")
               assert second.status_code == 401, f"post-deactivate refresh must 401, got {second.status_code}"
               second_code = (second.json() or {}).get("code") or (second.json() or {}).get("error")
               assert second_code in {"account_inactive", "invalid_token", "invalid_session"}, \
                   f"unexpected code {second_code}"
       else:
           assert refresh_status == 401, f"refresh race-loss expected 401, got {refresh_status}"
           code = refresh_body.get("code") or refresh_body.get("error")
           assert code in {"account_inactive", "invalid_token", "invalid_session"}, \
               f"unexpected code {code} (anti-oracle-safe set per Phase 43 Plan 43-10)"

       async with session_factory() as verify:
           is_active = (await verify.execute(text(
               "SELECT is_active FROM users WHERE id = :uid"
           ), {"uid": target_id})).scalar_one()
           assert is_active is False
   ```
