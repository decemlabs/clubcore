3. Seed + test body:
   ```python
   @pytest.mark.asyncio
   async def test_soft_delete_reinvite_race(
       real_commit_engine: AsyncEngine,
   ) -> None:
       session_factory = async_sessionmaker(real_commit_engine, expire_on_commit=False)

       shared_email = f"race-reinvite+{uuid4().hex[:8]}@local.dev"
       owner_id = uuid4()
       old_user_id = uuid4()

       # ── Seed owner (actor) + old user with shared_email ──
       async with session_factory() as setup:
           setup.add(User(id=owner_id, email=f"owner+{owner_id}@local.dev",
                          full_name="Race Owner", role="owner", is_active=True,
                          password_hash=hash_password("OwnerPass123!")))
           setup.add(User(id=old_user_id, email=shared_email,
                          full_name="Original Holder", role="reception", is_active=True,
                          password_hash=hash_password("OldPass123!")))
           await setup.commit()

       # ── Owner login → get session cookie + CSRF ──
       app = create_app()
       async with AsyncClient(transport=ASGITransport(app=app), base_url="http://test") as owner_client:
           login = await owner_client.post("/api/v1/auth/login",
                                            json={"email": f"owner+{owner_id}@local.dev", "password": "OwnerPass123!"})
           assert login.status_code == 200
           # owner_client now carries the auth cookie

           # ── Race: DELETE old user || POST new user with same email ──
           async def _delete_old() -> tuple[int, dict[str, Any]]:
               r = await owner_client.delete(f"/api/v1/users/{old_user_id}")
               return r.status_code, (r.json() if r.content else {})

           async def _create_new() -> tuple[int, dict[str, Any]]:
               r = await owner_client.post("/api/v1/users",
                                            json={"email": shared_email, "full_name": "New Holder", "role": "reception"})
               return r.status_code, (r.json() if r.content else {})

           (del_status, _), (post_status, _) = await asyncio.gather(_delete_old(), _create_new())

       # Either ordering is legitimate:
       # - DELETE first, POST after → (204|200, 201)
       # - POST first, DELETE after → (204|200, 409 Branch A)
       # The invariant is "exactly one active row with this email survives".
       assert del_status in {200, 204}, f"DELETE expected 200/204, got {del_status}"
       assert post_status in {201, 409}, f"POST expected 201 or 409, got {post_status}"

       async with session_factory() as verify:
           active_count = (await verify.execute(text(
               "SELECT count(*) FROM users WHERE lower(email) = lower(:e) AND deleted_at IS NULL"
           ), {"e": shared_email})).scalar_one()
           assert active_count == 1, f"expected exactly one active user with email {shared_email}, got {active_count}"

           old_row = (await verify.execute(text(
               "SELECT deleted_at FROM users WHERE id = :uid"
           ), {"uid": old_user_id})).first()
           assert old_row is not None
           assert old_row.deleted_at is not None, "old row must have deleted_at set"

           # If POST succeeded (201), assert the new row is distinct + active:
           if post_status == 201:
               new_row_count = (await verify.execute(text(
                   "SELECT count(*) FROM users WHERE lower(email) = lower(:e) AND id <> :old"
               ), {"e": shared_email, "old": old_user_id})).scalar_one()
               assert new_row_count == 1
   ```
