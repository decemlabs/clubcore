    from datetime import datetime, timedelta

    async def test_list_expiring_within_default_7(
        authed_client_owner: AsyncClient,
        make_plan: Any,
        make_membership: Any,
    ) -> None:
        """expiring=true with no `within` defaults to 7 days, inclusive end_date semantics
        (D-24-12 + PROJECT.md "Key Decisions"): end_date in [today, today+7] is INCLUDED.
        """
        today = datetime.now(ZoneInfo("Europe/Moscow")).date()
        plan = await make_plan(name="Default 7d")
        # In window — start of range:
        await make_membership(plan=plan, status="active",
                              start_date=today - timedelta(days=30), end_date=today)
        # In window — middle:
        await make_membership(plan=plan, status="active",
                              start_date=today - timedelta(days=30), end_date=today + timedelta(days=6))
        # In window — boundary at today+7 (inclusive end_date per D-24-12):
        await make_membership(plan=plan, status="active",
                              start_date=today - timedelta(days=30), end_date=today + timedelta(days=7))
        # Out of window — today+8:
        await make_membership(plan=plan, status="active",
                              start_date=today - timedelta(days=30), end_date=today + timedelta(days=8))

        resp = await authed_client_owner.get("/api/v1/memberships?expiring=true&pageSize=100")
        assert resp.status_code == 200
        body = resp.json()["data"]
        # Three rows in window (today, today+6, today+7); today+8 excluded.
        assert body["total"] == 3
        end_dates = sorted(item["endDate"] for item in body["items"])
        assert end_dates[0] == today.isoformat()
        assert end_dates[-1] == (today + timedelta(days=7)).isoformat()
        # Inclusive boundary at today+7 must appear:
        assert (today + timedelta(days=7)).isoformat() in end_dates
