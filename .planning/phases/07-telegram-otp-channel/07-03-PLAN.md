```python
class AppError(Exception):
    code: str = "app_error"
    status_code: int = 500
    def __init__(self, message: str, *, fields: dict[str, object] | None = None) -> None:
        self.message = message
        self.fields = fields
        super().__init__(message)

class CsrfMismatch(AppError):  # noqa: N818
    code = "csrf_mismatch"
    status_code = 403

class InvalidPassword(AppError):  # noqa: N818
    code = "invalid_credentials"
    status_code = 401

# ... (also: NotFoundError, ForbiddenError, ConflictError, ValidationAppError, InvalidAccessToken, RateLimited)

def register_exception_handlers(app: FastAPI) -> None:
    @app.exception_handler(AppError)
    async def _app_error_handler(request: Request, exc: AppError) -> JSONResponse:
        return JSONResponse(
            status_code=exc.status_code,
            content={"code": exc.code, "message": exc.message, "fields": exc.fields},
```

```python
"""Auth module-local exceptions — Phase 7 verify failure modes (D-13).

Эти классы живут в app.modules.auth (НЕ в app.core), потому что относятся
к узкому Telegram-OTP домену. Все подклассы наследуют app.core.exceptions.AppError;
это гарантирует, что существующий обработчик `_app_error_handler`
(app/core/exceptions.py:81-93) автоматически рендерит конверт
`{code, message, fields?}` без какой-либо дополнительной регистрации.

Соответствие табличке D-13:
  | Failure                                  | HTTP | code             | fields                |
  |------------------------------------------|------|------------------|-----------------------|
  | OtpCode found, code_hash IS NULL         | 409  | bot_not_started  | {deepLinkUrl}         |
  | expires_at < now                          | 410  | otp_expired      | —                     |
  | code_hash mismatch, attempts<5            | 401  | otp_invalid      | {attemptsRemaining}   |
  | code_hash mismatch, attempts>=5 after miss| 429  | otp_max_attempts | —                     |
  | consumed_at IS NOT NULL                   | 409  | otp_consumed     | —                     |
  | deep_link_token_hash not found            | 404  | token_unknown    | —                     |
"""

from app.core.exceptions import AppError


class BotNotStarted(AppError):  # noqa: N818
    """OtpCode найден, но `code_hash IS NULL` — пользователь не нажал /start у бота.

    Caller должен передавать `fields={"deepLinkUrl": <url>}` в `__init__`,
    чтобы FE мог повторно показать deep-link.
    """

    code = "bot_not_started"
    status_code = 409


class OtpExpired(AppError):  # noqa: N818
    """`expires_at < now` — TTL OTP-кода истёк (5 минут от выдачи)."""

    code = "otp_expired"
    status_code = 410


class OtpInvalid(AppError):  # noqa: N818
    """sha256(presented_code) != code_hash; attempts всё ещё < 5 после инкремента.

    Caller должен передавать `fields={"attemptsRemaining": n}` чтобы FE мог
    показать счётчик.
    """

    code = "otp_invalid"
    status_code = 401


class OtpMaxAttempts(AppError):  # noqa: N818
    """attempts >= 5 после очередного промаха — OTP заблокирован."""

    code = "otp_max_attempts"
    status_code = 429


class OtpAlreadyConsumed(AppError):  # noqa: N818
    """`consumed_at IS NOT NULL` — попытка повторного использования (replay)."""

    code = "otp_consumed"
    status_code = 409


class TokenUnknown(AppError):  # noqa: N818
    """Хеш deep_link_token не найден в OtpCode."""

    code = "token_unknown"
    status_code = 404
```

ВНИМАНИЕ: `# noqa: N818` обязателен на каждом классе (имена не оканчиваются на `Error` — ruff правило N818). Никаких `__init__` переопределений — наследуем от `AppError` as-is.
