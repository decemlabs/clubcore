import pytest
from httpx import AsyncClient
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.modules.auth.models import OtpCode


@pytest.mark.asyncio
async def test_telegram_start_returns_deep_link_and_creates_otp_row(
    async_client: AsyncClient,
    db_session: AsyncSession,

    """AUTH-TG-01: returns {deepLinkUrl, deepLinkToken} and creates OtpCode with code_hash IS NULL."""
    response = await async_client.post("/api/v1/auth/telegram/start")

    deep_link_url = body["data"]["deepLinkUrl"]

    assert len(deep_link_token) >= 32  # secrets.token_urlsafe(32) → 43 chars

    rows = (await db_session.execute(select(OtpCode))).scalars().all()

async def test_telegram_verify_happy_path(
    async_client: AsyncClient,
    db_session: AsyncSession,
    seeded_owner_with_tg: User,
    stub_telegram_sender,  # noqa: ANN001 — fixture
) -> None:
    """Full flow: /start → bot bind+commit (simulated direct call) → /verify → cookies + audit."""
    # 1. /start — get deep_link_token.
    r = await async_client.post("/api/v1/auth/telegram/start")
    assert r.status_code == 200
    deep_link_token = r.json()["data"]["deepLinkToken"]

    # 2. Simulate bot /start <token> handler under same SAVEPOINT (D-15: no live ptb).
    user, raw_code, otp_row = await telegram_service.bind_and_issue(
        db_session, deep_link_token, TG_CHAT_ID, TG_USERNAME,
    )
    # Sender returns ok=True by default (stub), so commit_otp is called.
    await telegram_service.commit_otp(db_session, otp_row, user, raw_code, TG_CHAT_ID)

    # 3. /verify with the raw code.
    with capture_logs() as caplog:
        verify = await async_client.post(
            "/api/v1/auth/telegram/verify",
            json={"deepLinkToken": deep_link_token, "code": raw_code},
        )
    assert verify.status_code == 200, verify.text

    # 4. Cookies present (AUTH-TG-04 — same shape as /login).
    set_cookies = verify.headers.get_list("set-cookie") if hasattr(verify.headers, "get_list") else verify.headers.raw
    joined = "\n".join(set_cookies) if isinstance(set_cookies, list) else "\n".join(str(c) for c in set_cookies)
    assert "sz_access=" in joined
    assert "sz_refresh=" in joined
    assert "sportzal_csrf=" in joined

    # 5. Audit events: otp_consumed + login_success(channel='telegram').
    events = [c.get("event") for c in caplog]
    assert "otp_consumed" in events, f"events={events}"
    assert "login_success" in events, f"events={events}"
    login_evt = next(c for c in caplog if c.get("event") == "login_success")
    assert login_evt.get("channel") == "telegram"

    # 6. Raw code never logged.
    for entry in caplog:
        for value in entry.values():
            assert raw_code not in str(value), f"raw OTP code leaked in event: {entry}"
